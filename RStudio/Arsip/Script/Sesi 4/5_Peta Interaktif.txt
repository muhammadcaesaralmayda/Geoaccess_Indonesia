# Packages Peta interaktif
library(leaflet)

# Mengatur crs data adalah WGS 84
adm_wgs <- st_transform(adm, 4326)
adm_wgs_2 <- st_zm(adm_wgs, drop = TRUE)
m <- leaflet(data=adm_wgs_2) %>% #Inisialisasi peta leaflet
  addTiles() %>% #basemap default OpenStreetMap
  addPolygons( #mengatur gambar poligon dari sf
    fillColor = "orange",
    fillOpacity = 0.1,
    color = "black",
    weight = 1
  )
m
#%>% artinya hasil sebelumnya dipakai sebagai input fungsi berikutnya
saveWidget(m, "Map_Bandung.html", selfcontained = TRUE) #R di Colab tidak bisa menampilkan langsung jadi perlu di download dulu
