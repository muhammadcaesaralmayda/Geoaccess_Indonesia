VISUALISASI DENGAN PLOT

# 1. Plot data vektor sederhana
library(sf)
plot(adm)              # plot lengkap (atribut + geometri)
plot(st_geometry(adm)) # hanya geometri (lebih bersih)


# 2. Membuat 1 plot data vektor dengan ditambah judul
plot( st_geometry(adm),main = "Wilayah Kota Bandung")


# 3. Menambah grid

# a. base R
plot((st_geometry(adm)),axes = TRUE) 
# TRUE menampilkan sumbu koordinat yang berfungsi sebagai grid referensi


# b. Mengatur interval grid
# Interval grid mengikuti satuan CRS (meter)
bb <- st_bbox(adm)

plot( st_geometry(adm), axes = FALSE, main = "Wilayah Kota Bandung")

axis(
  side = 1,
  at = seq(bb["xmin"], bb["xmax"], by = 0.1)
)

axis(
  side = 2,
  at = seq(bb["ymin"], bb["ymax"], by = 0.1)
)

box()


# 4. Menambah warna
plot((st_geometry(adm)),axes = TRUE, col="yellow", main = "PETA KOTA BANDUNG")
# declare warna dengan menambahkan col
# penulisan warna bisa dilakukan dengan 2 cara:
#1. dengan nama dalam bahasa inggris seperti yellow, red, blue, atau green.
#2. dengan kode hex. untuk kode hex perlu ditambahkan #sebelum kode dituliskan.


# 5. Menambah background
plot((st_geometry(adm)),axes = TRUE, col="yellow", main = "PETA KOTA BANDUNG", bg = "#0099ff")
#declare background dengan bg

# 7. Membuat 1 peta dengan beberapa overlay layer

# a. Menambah shp baru
shp1 <- st_read("PEMUKIMAN_AR_25K.shp")
shp2 <- st_transform(shp1, 32748)

# b. Menambah shp baru
shp3 <- st_read ("Jalan.shp")
shp4 <- st_transform(shp3, 32748)

# c. plotting
plot(st_geometry(shp2),
     col = "yellow", # warna
     axes = TRUE, # grid
     asp = 1, # untuk atur rasio
     main = "PETA KOTA BANDUNG" # judul
)

plot(st_geometry(shp4),
     col = "pink",
     asp = 1,
     add = TRUE, #untuk tambah layer
     border = NA
)

# 8. Mengarahkan zoom ke Area of Interest (AoI)

# a. Menambahkan fungsi untuk zoom di aoi
shp5 <- st_read("Jalan_Regol.shp")
st_crs(shp5)
shp6 <- st_transform(shp5,32748)
zoom<- st_bbox (shp6) #area of interes

# b. Membuat Plot Peta
plot(st_geometry(shp2), #background
     col = "yellow",
     axes = TRUE,
     asp = 1,
     main = "PETA KOTA BANDUNG",
     xlim =c(zoom["xmin"], zoom["xmax"]), #zoom to layer x dari AoI
     ylim =c(zoom["ymin"], zoom["ymax"]), #zoom to layer y dari AoI
     border = NA
     )

plot(st_geometry(shp4), #building
     col = "pink",
     asp = 1,
     add = TRUE,
     border = NA
)

plot(st_geometry(shp6), #road
     col = "green",
     asp = 1,
     add = TRUE,
     border = NA
)
