# 1. pastikan package sudah ada
library(ggspatial)

# 2. Plot Peta sederhana
ggplot() +
  geom_sf(data = adm, aes(fill = PENDUDUK))+
  ggtitle("Peta Choropleth")


# 3. Beberapa warna tapi hanya : kuning - hijau - biru - ungu
ggplot() +
  geom_sf(data = adm, aes(fill = PENDUDUK))+
  scale_fill_viridis_c(
    name = "Jumlah Penduduk",
  )+
ggtitle("Peta Choropleth Kuning - Ungu")


# 4. Mengatur agar warna lebih flexible
ggplot(adm) +
  geom_sf(aes(fill = PENDUDUK), color = "white", linewidth = 0.2) +
  scale_fill_gradient(
    name = "Jumlah Penduduk",
    low = "navy",
    high = "red",
  ) +
  ggtitle("Choropleth Penduduk per Kelurahan") +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold")
  )


# 5. Mengatur warna dengan palette Set 3
ggplot(shp) +
  geom_sf(aes(fill = KOTA), color = "white", linewidth = 0.2) +
  scale_fill_brewer(palette = "Set3",
                    name = "Jumlah Penduduk",
                    
  ) +
  ggtitle("Choropleth Penduduk per Kelurahan") +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold")
  )

# 6. Banyak warna
ggplot(adm) +
  geom_sf(aes(fill = WADMKC), color = "white", linewidth = 0.2) +
  scale_fill_hue(name = "Kecamatan") +
  ggtitle("Choropleth per Kecamatan") +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold")
  )
