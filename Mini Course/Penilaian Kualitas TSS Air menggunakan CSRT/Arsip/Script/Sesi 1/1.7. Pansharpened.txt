//******************** Image Mosaic & Cloud Masking ********************

var landsat9_mask = function(image) {

  // Bitmask untuk bayangan awan
  var cloudShadowBitMask = (1 << 3);

  // Bitmask untuk awan
  var cloudsBitMask = (1 << 5);

  //Memilih band QA_PIXEL dari citra
  var qa = image.select('QA_PIXEL');

  // Piksel tanpa bayangan awan
  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)

    // Piksel tanpa awan
    .and(qa.bitwiseAnd(cloudsBitMask).eq(0));
    return image.updateMask(mask);
};


//******************** Image Pansharpened ********************

//Metode HSV (Hue, Saturation, Value) untuk pansharpening
var pansharpening = function(image) {
  var rgb = image.select('B4', 'B3', 'B2');
  var pan = image.select('B8');
  
//Convert RGB to HSV, swap in the pan band, and convert back to RGB.
  var huesat = rgb.rgbToHsv().select('hue', 'saturation');
  var upres = ee.Image.cat(huesat, pan).hsvToRgb();
  return image.addBands(upres);
};


//******************** Filter, Load, & Clip Image********************

//Koleksi Citra
var landsat9 = ee.ImageCollection('LANDSAT/LC09/C02/T1_TOA')

  //Filter Tanggal
  .filterDate('2022-01-01', '2023-01-01')

  //Load Masking & Pansharpened Citra
  .map(landsat9_mask)
  .map(pansharpening)
  .median()

  //Clip Raster
  .clip(geometry)

  //Multiple Band 
  .select (['red', 'green', 'blue']);

//Parameter Visualisasi Multipleband
var landsat9_param = {min: 0, max: 0.3,};

//Menampilkan Citra pada Layer Peta
Map.addLayer(landsat9, landsat9_param, 'Pansharpened');

//Zoom ke ROI
Map.centerObject(geometry,12);
