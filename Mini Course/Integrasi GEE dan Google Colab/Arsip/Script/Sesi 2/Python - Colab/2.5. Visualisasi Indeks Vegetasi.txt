fig, axes = plt.subplots(1, 4, figsize=(20, 5))

# =========================
# 1. NDVI
# =========================
im1 = axes[0].imshow(
    indices[:, :, 0],
    vmin=-1,
    vmax=1,
    cmap='RdYlGn'
)
axes[0].set_title('NDVI')
axes[0].axis('off')

cbar1 = fig.colorbar(
    im1,
    ax=axes[0],
    fraction=0.046,

)
