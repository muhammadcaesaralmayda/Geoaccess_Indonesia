// script kembali ke menampilkan citra lagi

rgb_stretch = np.zeros_like(rgb)

for i in range(3):
    p2 = np.nanpercentile(rgb[:, :, i], 2)
    p98 = np.nanpercentile(rgb[:, :, i], 98)

    rgb_stretch[:, :, i] = np.clip(
        (rgb[:, :, i] - p2) / (p98 - p2),
        0, 1
    )

plt.figure(figsize=(6, 6))
plt.imshow(rgb_stretch)
plt.title('True Color\n(B4-B3-B2)')
plt.axis('off')
plt.show()
