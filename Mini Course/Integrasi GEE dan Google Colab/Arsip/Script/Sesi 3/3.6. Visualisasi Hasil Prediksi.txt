# Visualisasi ditambah dengan RGB

s2_rgb = geemap.ee_to_numpy(
    predictors.select(['B4', 'B3', 'B2']),
    region=predictors.geometry(),
    scale=100
)

# Stretch setiap band ke 0-1
rgb_stretch = np.zeros_like(s2_rgb)

for i in range(3):
    band = s2_rgb[:, :, i]

    band_min = np.nanmin(band)
    band_max = np.nanmax(band)

    rgb_stretch[:, :, i] = (
        (band - band_min) /
        (band_max - band_min)
    )

fig, axes = plt.subplots(1, 2, figsize=(16, 7))

# Sentinel-2
axes[0].imshow(rgb_stretch)
axes[0].set_title(
    "Sentinel-2 Multispectral\n(B4 - B3 - B2)"
)
axes[0].axis("off")
