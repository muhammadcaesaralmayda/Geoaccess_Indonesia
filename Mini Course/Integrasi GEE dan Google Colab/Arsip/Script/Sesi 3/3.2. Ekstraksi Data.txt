# mengubah data kanopi menjadi ee feature
""" Hal ini perlu karena sampleRegions() di GEE membutuhkan sekumpulan feature/titik referensi sebagai collection."""

import ee
import json

features = []

for _, row in df.iterrows():

    geojson = json.loads(row['.geo'])

    # Ambil koordinat dari GeoJSON
    coordinates = geojson['coordinates']

    # Buat titik Earth Engine
    point = ee.Geometry.Point(coordinates)

    # Buat feature + target canopy cover
    feature = ee.Feature(
        point,
        {
            'canopy_cover': float(row['canopy_cover'])
        }
    )

    features.append(feature)

reference = ee.FeatureCollection(features)

print(
    "Jumlah titik reference:",
    reference.size().getInfo()
)

#------------------------------------------------------------------

# Ekstraksi
samples = predictors.sampleRegions(
    collection=reference,
    properties=['canopy_cover'],
    scale=10,
    geometries=True
)
