//************************ Load Image Singleband ************************

// Load Citra
var sentinel2 = ee.Image('COPERNICUS/S2_SR_HARMONIZED/20240815T023529_20240815T025522_T49MDN')

  // Single Band 
  .select('B8')

  // Faktor Skala Band
  .multiply(0.0001);

// Parameter Visualisasi Singleband 
var sentinel2_param = {
  min: 0, 
  max: 0.45, 
  palette: ['black', 'white']
};

// Menampilkan Citra pada Layer Peta
Map.addLayer(sentinel2, sentinel2_param, 'B5 NIR');

// Zoom ke ROI
Map.centerObject(geometry, 10);
