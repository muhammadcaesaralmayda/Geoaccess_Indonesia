******************************************** NORMALISASI RENTANG NILAI PIKSEL **********************************************

# Buat fungsi untuk melakukan Normalisasi
def normalize(array):
    array_min, array_max = array.min(), array.max()
    return ((array - array_min)/(array_max - array_min))

# (x - max)/(max-min)

# Normalisasi band
red_n = normalize(red)
green_n = normalize(green)
blue_n = normalize(blue)

# Gabungkan jadi RGB rgb
rgb = np.dstack((red_n, green_n, blue_n))

# Tampilkan RGB
plt.imshow(rgb)


#Cek nilai min-max 
red_n.min(), red_n.max()

******************************************** ENHANCED **********************************************
def enhance(band):
    p2, p98 = np.percentile(band, (2, 98))
    return np.clip((band - p2) / (p98 - p2), 0, 1)

red = enhance(red)
green = enhance(green)
blue = enhance(blue)

rgb_enhanced = np.dstack((red, green, blue))

plt.imshow(rgb_enhanced)
plt.title("Enhanced RGB per Band")
plt.axis("off")
plt.show()
