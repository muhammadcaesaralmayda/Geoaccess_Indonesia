# Membuat contoh axes
fig, ax = plt.subplots(nrows=2,

#Contoh memberikan judul
fig, ax = plt.subplots(figsize=(10,10))
ax.set_title('Contoh Judul', fontsize=8)

#Contoh setting nilai min-max sumbu x dan ya
ax.set_xlim(0.0, 2.0)
ax.set_ylim(0.0, 2.0)

plt.show()

************************************************** MENAMPILKAN MULTIPLE AXES **********************************************************

# Membuat 4 axes
fig, ax = plt.subplots(nrows=2, ncols=2, figsize=(7,7))

# Beri judul besar
fig.suptitle('4 Axes', color='blue')

# beri judul pasa masing-masing axes
ax[0, 0].set_title('Axes 1', color='red')
ax[0, 1].set_title('Axes 2')
ax[1, 0].set_title('Axes 3')
ax[1, 1].set_title('Axes 4')

plt.show()
