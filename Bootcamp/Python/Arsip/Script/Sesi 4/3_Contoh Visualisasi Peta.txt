# Membuat peta pada dua axes

# Membuat peta dengan warna dan menambahkan judul
fig, ax = plt.subplots(1, 2, figsize=(15, 10))  # Mengatur ukuran peta

# Peta terang
adm_bdg.plot(ax=ax[0], edgecolor='black', facecolor='lightblue')
ax[0].set_title("Peta Administrasi Kota Bandung (Terang)", fontsize=15)

# Peta gelap
adm_bdg.plot(ax=ax[1], edgecolor='#222222', facecolor='white')
ax[1].set_title("Peta Administrasi Kota Bandung (Gelap)", fontsize=15)
ax[1].set_facecolor('#222222')

# Menampilkan peta
plt.show()
