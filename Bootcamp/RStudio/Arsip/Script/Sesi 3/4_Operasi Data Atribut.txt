# 13. Transformasi Koordinat

# a. load data
library(sf)
pemukiman 	<- st_read("PEMUKIMAN_AR_25K.shp")
jalan 		<- st_read("Jalan.shp")
pendidikan	<- st_read("PENDIDIKAN_PT_25K.shp")


# b. cek CRS sekarang
st_crs(pemukiman)
st_crs(jalan)
st_crs(pendidikan)

# c. ubah CRS ke WGS84
pemukiman_UTM <- st_transform(pemukiman, 32748)
jalan_UTM <- st_transform(jalan, 32748)
pendidikan_UTM <- st_transform(pendidikan, 32748)

# d. cek hasil
st_crs(pemukiman_UTM)
st_crs(jalan_UTM)
st_crs(pendidikan_UTM)
