# 1. Alamat Bangunan
pemukiman_adm <- st_join(
  pemukiman_utm, #objek 1
  adm_utm,       # objek 2
  join = st_within 
)

# 2. Alamat Jalan
jalan_admin <- st_join(
  jalan_utm,
  adm_utm,
  join = st_intersects
)
