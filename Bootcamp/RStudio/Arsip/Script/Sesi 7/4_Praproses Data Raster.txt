# 2. Menyamakan system referensi koordinat

# a. Baca data
L8  <- rast("L8_2020_YOGYAKARTA.tif")
adm <- vect("Batas_Adm_Yogyakarta.shp")

# b. Cek CRS masing-masing
crs(L8)
crs(adm)

# c. Samakan CRS (vektor mengikuti raster)
adm_crs <- project(adm, crs(L8))

-------------------------------------------------------------------------------------------

# 3. Crop + mask (zoom dulu, lalu dirapikan)
L8_crop <- crop(L8, adm_crs) #zoom ke arah xy max dan min
L8_mask <- mask(L8_crop, adm_crs)

-------------------------------------------------------------------------------------------

# 4. Plot peta
plot (L8_mask,1)
