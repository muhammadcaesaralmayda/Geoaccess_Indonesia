OPERASI MATEMATIS

library(terra)
# 1. Ambil band (Landsat 8)
# Red = band 4, NIR = band 5
red <- L8_mask[[4]]
nir <- L8_mask[[5]]

-------------------------------------------------------------------------------------------

# 2. Hitung NDVI
ndvi <- (nir - red) / (nir + red)

-------------------------------------------------------------------------------------------

# 3. Cek nilai NDVI
global(ndvi, c("min", "max"), na.rm = TRUE)

-------------------------------------------------------------------------------------------

# 4. Visualisasi
plot(
  ndvi,
  col  = terrain.colors(100),
  main = "NDVI Yogyakarta"
)

-------------------------------------------------------------------------------------------

# 5. Save as
writeRaster(
  ndvi,
  "NDVI_KOTA JOGJA.tif",
  overwrite = TRUE
)
