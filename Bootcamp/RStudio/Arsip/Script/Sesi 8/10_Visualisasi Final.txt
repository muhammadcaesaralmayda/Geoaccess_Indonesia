# 23. MENGHITUNG TOTAL LUAS PERUBAHAN VEGETASI
perubahan_luas_vegetasi <- vegetasi %>%
  
  st_drop_geometry() %>%
  
  group_by(Perubahan_Vegetasi) %>%
  
  summarise(
    Luas_Ha = sum(Luas_Ha),
    .groups = "drop"
  )


# Lihat hasil
perubahan_luas_vegetasi
