//************************Load Image Singleband************************

// Load Citra
var landsat9 = ee.Image('LANDSAT/LC09/C02/T1_L2/LC09_120065_20220727')

  // Single Band 
  .select('SR_B5')

  // Faktor Skala Band
  .multiply(0.0000275).add(-0.2);

// Parameter Visualisasi Singleband 
var landsat9_param = {
  min: 0, 
  max: 0.3, 
  palette: ['black', 'white']
};

// Menampilkan Citra pada Layer Peta
Map.addLayer(landsat9, landsat9_param, 'SR_B5');

// Zoom ke ROI
Map.centerObject(geometry);

  // Multiple Band
