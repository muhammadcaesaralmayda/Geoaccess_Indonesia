// *********************** Panel ************************************

// Menampilkan Dataset Rata-Rata Curah Hujan Tahunan pada Layer Peta
var mapPanel = ui.Map();
mapPanel.addLayer(tahun2022Clip, vis, 'Mean Annual Precipitation');
mapPanel.centerObject(geometry, 8);

// Panel User Interface
var inspectorPanel = ui.Panel({
  style: {
    position: 'top-left',
    width: '300px',
    padding: '8px'
  }
});

// Panel Judul dan Deskripsi
var intro = ui.Panel([
  ui.Label({
    value: 'Analisis Curah Hujan Tahunan',
    style: {fontSize: '20px', fontWeight: 'bold'}
  }),
  ui.Label({
    value: 'Peta ini menampilkan distribusi rata-rata curah hujan tahunan dalam wilayah tertentu berdasarkan data dari koleksi CHIRPS (Climate Hazards Group InfraRed Precipitation with Station Data). Data curah hujan dikalkulasi untuk periode waktu 2000–2023 dengan menggunakan metode penghitungan tahunan dan dirata-rata untuk menghasilkan gambaran umum pola curah hujan dalam jangka panjang.',
    style: {fontSize: '13px', textAlign: 'justify'}
  })
]);
inspectorPanel.add(intro);

// Legenda Warna
function makeColorBarParams(palette) {
  return {
    bbox: [0, 0, 1, 0.1],
    dimensions: '100x10',
    format: 'png',
    min: 0,
    max: 1,
    palette: palette,
  };
}

// Membuat Legenda Warna
var legend = ui.Thumbnail({
  image: ee.Image.pixelLonLat().select(0),
  params: makeColorBarParams(vis.palette),
  style: {stretch: 'horizontal', margin: '0px 8px', maxHeight: '24px'}
});

// Panel untuk Legenda Warna
var legendLabels = ui.Panel({
  widgets: [
    ui.Label(vis.min, {margin: '4px 8px'}),
    ui.Label((vis.max / 2), {margin: '4px 8px', textAlign: 'center', stretch: 'horizontal'}),
    ui.Label(vis.max, {margin: '4px 8px'})
  ],
  layout: ui.Panel.Layout.flow('horizontal')
});

// Membuat Judul Legenda
var legendTitle = ui.Label({
  value: 'Legenda Curah Hujan (mm)',
  style: {fontWeight: 'bold'}
});

// Menggabungkan Panel Legenda
var legendPanel = ui.Panel([legendTitle, legend, legendLabels]);
inspectorPanel.add(legendPanel);

// Sumber Data dan URL 
var sumberLabel = ui.Label({
  value: 'Sumber: CHIRPS Daily',
  style: {fontWeight: 'bold', color: 'black'},
  targetUrl: 'https://developers.google.com/earth-engine/datasets/catalog/UCSB-CHG_CHIRPS_DAILY'
});
inspectorPanel.add(sumberLabel);

// Tambahkan Panel ke UI
ui.root.clear();
mapPanel.add(inspectorPanel);
ui.root.add(mapPanel);
